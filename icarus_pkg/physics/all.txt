"""

Stage 1 Two-Pass Global Pseudo-Continuum Cascade (AGN Power-Law + Host Starlight +
Balmer Edge + FFT Fe II Deconvolution) and Stage 2 Local Boundary-Condition Pinning.
"""

from typing import List, Optional, Tuple

import numpy as np
from astropy.stats import sigma_clipped_stats
from lmfit import Model
from lmfit.models import ConstantModel, LinearModel, PolynomialModel, QuadraticModel

from ..core.config import FittingConfig

log = get_icarus_logger()
PIVOT_WAVE_AA = 5100.0
BALMER_EDGE_AA = 3646.0


def _agn_powerlaw_host_balmer(
    x: np.ndarray,
    pl_amp: float = 1.0,
    pl_alpha: float = -1.5,
    host_amp: float = 0.0,
    balmer_amp: float = 0.0,
) -> np.ndarray:
    """
    Roadmap Eq. (1): AGN non-thermal power-law diluted by host starlight plus an
    optional optically thick Balmer continuum edge below 3646 A ("Small Blue Bump").
    """
    w_safe = np.maximum(np.asarray(x, dtype=np.float64), 100.0)
    pl_flux = pl_amp * ((w_safe / PIVOT_WAVE_AA) ** pl_alpha)

    # Smooth host galaxy starlight component (peaking in rest-frame optical/NIR)
    host_flux = (
        host_amp * ((w_safe / PIVOT_WAVE_AA) ** 0.35) * np.exp(-w_safe / 25000.0)
    )

    # Optically thick Balmer continuum below 3646 A (Grandi 1982 approximation)
    # h*c / (k_B * T_e) at T_e = 15,000 K is (1.4388e8 K A / 15000 K) = 9592.0 A
    balmer_flux = np.zeros_like(w_safe)
    if balmer_amp > 0.0:
        uv_mask = w_safe <= BALMER_EDGE_AA
        if np.any(uv_mask):
            w_uv = w_safe[uv_mask]
            ratio = w_uv / BALMER_EDGE_AA
            exponent = -(1.4388e8 / 15000.0) * (1.0 / w_uv - 1.0 / BALMER_EDGE_AA)
            balmer_flux[uv_mask] = (
                balmer_amp * (ratio**3) * np.exp(np.clip(exponent, -50.0, 0.0))
            )

    return pl_flux + host_flux + balmer_flux


class ContinuumFitter:
    """Executes Stage 1 global SED + Fe II fitting and Stage 2 local boundary pinning."""

    _FEII_CONVOLVER: Optional[FeIITemplateConvolver] = None

    @classmethod
    def _get_feii_convolver(cls) -> FeIITemplateConvolver:
        if cls._FEII_CONVOLVER is None:
            cls._FEII_CONVOLVER = FeIITemplateConvolver()
        return cls._FEII_CONVOLVER

    @staticmethod
    @dev_hook(step_name="build_polynomial_continuum")
    def build_polynomial_continuum(
        degree: int = 2, prefix: str = "cont_", flux_median: float = 1.0
    ) -> SpectralComponent:
        """Constructs a bounded polynomial continuum component (degrees 0-3)."""
        safe_median = abs(float(flux_median)) if abs(flux_median) > 1e-5 else 1e-5

        if degree == 0:
            model = ConstantModel(prefix=prefix)
            params = model.make_params(
                c=dict(value=flux_median, min=-safe_median * 5, max=safe_median * 5)
            )
        elif degree == 1:
            model = LinearModel(prefix=prefix)
            params = model.make_params(
                slope=0.0,
                intercept=dict(
                    value=flux_median, min=-safe_median * 5, max=safe_median * 5
                ),
            )
        elif degree == 2:
            model = QuadraticModel(prefix=prefix)
            params = model.make_params(
                a=0.0,
                b=0.0,
                c=dict(value=flux_median, min=-safe_median * 5, max=safe_median * 5),
            )
        elif degree == 3:
            model = PolynomialModel(degree=3, prefix=prefix)
            params = model.make_params(
                c0=dict(value=flux_median, min=-safe_median * 5, max=safe_median * 5),
                c1=0.0,
                c2=0.0,
                c3=0.0,
            )
        else:
            raise IcarusKinematicError("Continuum degree must be between 0 and 3.")

        return SpectralComponent(
            name=prefix.strip("_"),
            model=model,
            params=params,
            component_type="continuum",
        )

    @classmethod
    @dev_hook(step_name="fit_global_sed")
    def fit_global_sed(
        cls, spectrum: Spectrum1D, config: FittingConfig
    ) -> Tuple[Optional[SpectralComponent], Optional[np.ndarray], Optional[np.ndarray]]:
        """
        Executes the Stage 1 Two-Pass Global Pseudo-Continuum Cascade:
          Pass 1: Fits AGN power-law + host starlight (or polynomial) in Fe-free windows.
          Pass 2: Tightly bounds Pass 1 parameters (alpha +/- 0.15, amp +/- 15%) and
                  co-fits UV/Optical Fe II templates across `feii_windows`.
        Returns: (continuum_component, continuum_eval_array, feii_eval_array)
        """
        DevState.emit("pre_stage1_continuum", locals())

        wave_full, flux_full, err_full = spectrum.to_ndarrays()
        valid_base = (~spectrum.mask) & np.isfinite(wave_full) & np.isfinite(flux_full)

        cont_mask = np.zeros_like(wave_full, dtype=bool)
        populated_windows = 0
        for window in config.continuum_windows:
            w_in = valid_base & (wave_full >= window[0]) & (wave_full <= window[1])
            if np.sum(w_in) >= 3:
                populated_windows += 1
            cont_mask |= w_in

        feii_eval = np.zeros_like(wave_full, dtype=np.float64)

        if np.sum(cont_mask) < 10 or populated_windows < 2:
            log.info(
                f"[{spectrum.target.target_id}] Sparse global anchor coverage ({populated_windows} windows). Using local boundary-condition continuum."
            )
            return None, None, feii_eval

        x_cont = wave_full[cont_mask]
        y_cont = flux_full[cont_mask]
        err_cont = err_full[cont_mask]
        weights_cont = 1.0 / np.maximum(err_cont, 1e-5)
        median_flux = max(float(np.nanmedian(y_cont)), 1e-4)

        # --- PASS 1: Pure Continuum Anchor Fit ---
        global_cont_eval: Optional[np.ndarray] = None
        global_cont_comp: Optional[SpectralComponent] = None
        used_powerlaw = False

        if config.cont_type in ("powerlaw_feii", "powerlaw"):
            pl_model = Model(_agn_powerlaw_host_balmer, prefix="sed_")
            pl_params = pl_model.make_params()

            w_med = float(np.nanmedian(x_cont))
            amp_init = max(median_flux * ((w_med / PIVOT_WAVE_AA) ** 1.5), 1e-4)

            pl_params["sed_pl_amp"].set(
                value=amp_init, min=0.0, max=max(median_flux * 50.0, 10.0)
            )
            pl_params["sed_pl_alpha"].set(value=-1.5, min=-3.5, max=1.5)
            pl_params["sed_host_amp"].set(
                value=median_flux * 0.2, min=0.0, max=max(median_flux * 10.0, 5.0)
            )

            has_balmer_edge = np.min(x_cont) < 3600.0 and np.max(x_cont) > 3700.0
            if has_balmer_edge:
                pl_params["sed_balmer_amp"].set(
                    value=median_flux * 0.05, min=0.0, max=median_flux * 3.0
                )
            else:
                pl_params["sed_balmer_amp"].set(value=0.0, vary=False)

            try:
                res_p1 = pl_model.fit(y_cont, pl_params, x=x_cont, weights=weights_cont)
                if res_p1.success:
                    global_cont_eval = res_p1.eval(x=wave_full)
                    global_cont_comp = SpectralComponent(
                        name="sed",
                        model=pl_model,
                        params=res_p1.params.copy(),
                        component_type="continuum",
                    )
                    used_powerlaw = True
            except Exception as e:
                log.debug(
                    f"[{spectrum.target.target_id}] Power-law fit fallback triggered: {e}"
                )

        # Fallback to polynomial if power-law was not requested or did not converge
        if global_cont_eval is None:
            poly_comp = cls.build_polynomial_continuum(
                degree=config.continuum_degree, flux_median=median_flux
            )
            try:
                res_poly = poly_comp.model.fit(
                    y_cont, poly_comp.params, x=x_cont, weights=weights_cont
                )
                global_cont_eval = res_poly.eval(x=wave_full)
                global_cont_comp = SpectralComponent(
                    name="cont",
                    model=poly_comp.model,
                    params=res_poly.params.copy(),
                    component_type="continuum",
                )
            except Exception as e:
                log.warning(
                    f"[{spectrum.target.target_id}] Stage 1 global continuum warning: {e}"
                )
                return None, None, feii_eval

        # --- PASS 2: Empirical Fe II Pseudo-Continuum Co-Fit ---
        if (
            config.enable_feii
            and config.cont_type == "powerlaw_feii"
            and global_cont_comp is not None
        ):
            feii_mask = np.zeros_like(wave_full, dtype=bool)
            for fw in config.feii_windows:
                feii_mask |= valid_base & (wave_full >= fw[0]) & (wave_full <= fw[1])

            if np.sum(feii_mask) >= 8:
                y_fe_res = flux_full[feii_mask] - global_cont_eval[feii_mask]
                e_fe = err_full[feii_mask]
                pos_res = y_fe_res[y_fe_res > 0]
                fe_guess = (
                    float(np.nanpercentile(pos_res, 85)) if len(pos_res) > 2 else 0.0
                )

                if fe_guess > 0.5 * float(np.nanmedian(e_fe)):
                    convolver = cls._get_feii_convolver()
                    fe_comp = convolver.build_lmfit_model(
                        uv_amp_guess=fe_guess,
                        opt_amp_guess=fe_guess,
                        max_amp=max(median_flux * 5.0, 1.0),
                    )

                    x_fe_only = wave_full[feii_mask]
                    if np.min(x_fe_only) > 3500.0:
                        fe_comp.params["feii_uv_amp"].set(value=0.0, vary=False)
                    if np.max(x_fe_only) < 3500.0:
                        fe_comp.params["feii_opt_amp"].set(value=0.0, vary=False)

                    try:
                        if used_powerlaw:
                            # Tightly bound Pass 1 power-law parameters (alpha +/- 0.15, amp +/- 15%)
                            # and co-fit across the combined continuum + Fe II anchor windows
                            joint_mask = cont_mask | feii_mask
                            x_joint = wave_full[joint_mask]
                            y_joint = flux_full[joint_mask]
                            w_joint = 1.0 / np.maximum(err_full[joint_mask], 1e-5)

                            co_model = global_cont_comp.model + fe_comp.model
                            co_params = global_cont_comp.params.copy()
                            co_params.update(fe_comp.params)

                            p1_amp = float(co_params["sed_pl_amp"].value)
                            p1_alpha = float(co_params["sed_pl_alpha"].value)
                            co_params["sed_pl_amp"].set(
                                value=p1_amp,
                                min=max(0.0, p1_amp * 0.85),
                                max=max(p1_amp * 1.15, 1e-4),
                            )
                            co_params["sed_pl_alpha"].set(
                                value=p1_alpha,
                                min=max(-3.5, p1_alpha - 0.15),
                                max=min(1.5, p1_alpha + 0.15),
                            )
                            co_params["sed_host_amp"].set(vary=False)
                            co_params["sed_balmer_amp"].set(vary=False)

                            co_res = co_model.fit(
                                y_joint, co_params, x=x_joint, weights=w_joint
                            )
                            if co_res.success:
                                global_cont_eval = global_cont_comp.model.eval(
                                    co_res.params, x=wave_full
                                )
                                feii_eval = fe_comp.model.eval(
                                    co_res.params, x=wave_full
                                )
                                for p_key in global_cont_comp.params:
                                    if p_key in co_res.params:
                                        global_cont_comp.params[p_key].value = (
                                            co_res.params[p_key].value
                                        )
                        else:
                            w_fe = 1.0 / np.maximum(e_fe, 1e-5)
                            fe_res = fe_comp.model.fit(
                                y_fe_res, fe_comp.params, x=x_fe_only, weights=w_fe
                            )
                            if fe_res.success:
                                feii_eval = fe_res.eval(x=wave_full)
                    except Exception as e:
                        log.debug(
                            f"[{spectrum.target.target_id}] Fe II template co-fit skipped: {e}"
                        )

        DevState.emit("post_stage1_continuum", locals())
        return global_cont_comp, global_cont_eval, feii_eval

    @staticmethod
    @dev_hook(step_name="build_local_continuum")
    def build_local_boundary_continuum(
        sliced_spec: Spectrum1D,
        lines: List[EmissionLineDef],
        prefix: str = "cont_",
    ) -> SpectralComponent:
        """
        Stage 2 Local Boundary-Condition Pinning:
        Constructs a local linear continuum anchored directly to the left and right
        line-free boundary edges (25% margins) of the sliced complex window so
        Gaussian wings strictly obey local data boundary conditions.
        """
        x, y, err = sliced_spec.to_ndarrays()
        valid = (~sliced_spec.mask) & np.isfinite(x) & np.isfinite(y)
        x_val, y_val, err_val = x[valid], y[valid], err[valid]

        line_free_mask = np.ones_like(x_val, dtype=bool)
        for line in lines:
            line_free_mask &= np.abs(x_val - line.rest_wavelength) > 25.0

        if np.sum(line_free_mask) >= 6:
            x_lf = x_val[line_free_mask]
            y_lf = y_val[line_free_mask]
            e_lf = err_val[line_free_mask]
        else:
            x_lf, y_lf, e_lf = x_val, y_val, err_val

        n_pts = len(x_lf)
        edge_n = max(2, int(n_pts * 0.25))

        x_left = float(np.nanmedian(x_lf[:edge_n]))
        _, y_left, std_left = sigma_clipped_stats(y_lf[:edge_n], sigma=2.5, maxiters=3)
        x_right = float(np.nanmedian(x_lf[-edge_n:]))
        _, y_right, std_right = sigma_clipped_stats(
            y_lf[-edge_n:], sigma=2.5, maxiters=3
        )

        local_rms = max(float(np.nanmedian([std_left, std_right])), 1e-4)

        model = LinearModel(prefix=prefix)
        if abs(x_right - x_left) > 5.0:
            slope_init = (y_right - y_left) / (x_right - x_left)
            intercept_init = y_left - slope_init * x_left

            params = model.make_params(slope=slope_init, intercept=intercept_init)
            try:
                weights = 1.0 / np.maximum(e_lf, 1e-5)
                pre_res = model.fit(y_lf, params, x=x_lf, weights=weights)
                slope_init = float(pre_res.params[f"{prefix}slope"].value)
                intercept_init = float(pre_res.params[f"{prefix}intercept"].value)
            except Exception:
                pass

            slope_tol = max(
                abs(slope_init) * 0.25, (2.0 * local_rms) / abs(x_right - x_left)
            )
            int_tol = max(abs(intercept_init) * 0.15, 2.0 * local_rms)

            params[f"{prefix}slope"].set(
                value=slope_init,
                min=slope_init - slope_tol,
                max=slope_init + slope_tol,
            )
            params[f"{prefix}intercept"].set(
                value=intercept_init,
                min=intercept_init - int_tol,
                max=intercept_init + int_tol,
            )
        else:
            _, med_y, _ = sigma_clipped_stats(y_lf, sigma=2.5, maxiters=3)
            safe_med = max(abs(float(med_y)), 1e-4)
            params = model.make_params(
                slope=dict(value=0.0, vary=False),
                intercept=dict(
                    value=float(med_y),
                    min=float(med_y) - 2.0 * safe_med,
                    max=float(med_y) + 2.0 * safe_med,
                ),
            )

        return SpectralComponent(
            name=prefix.strip("_"),
            model=model,
            params=params,
            component_type="continuum",
        )
"""
demographics.py

Single-Epoch Virial Black Hole Mass Calibrations, Coatman et al. (2017) C IV
Blueshift Corrections, Accretion Telemetry (L_bol, lambda_Edd, R_FeII),
BPT Diagnostics, Star Formation Rates, and 6-Bit Confidence Flag Evaluation
for ICARUS v3.4.
"""

__project__ = "ICARUS"
__author__ = "Joseph Havens"
__version__ = "3.4.0"

from typing import Any, Dict, List, Optional, Tuple

import numpy as np
from astropy import constants as const
from astropy import units as u

from ..core.config import IcarusConfig
from ..core.containers import Spectrum1D
from ..core.dev_tools import DevState, dev_hook
from .line_models import UV_WIND_IONS

C_KMS = const.c.to("km/s").value
MPC_TO_CM = 3.08567758e24
FWHM_FACTOR = 2.0 * np.sqrt(2.0 * np.log(2.0))
_TRAPZ = getattr(np, "trapezoid", getattr(np, "trapz", None))


class DemographicsCalculator:
    """Computes deconvolved AGN virial demographics, BPT ratios, and confidence bitmasks."""

    @staticmethod
    def _get_luminosity_distance_cm(redshift: float, config: IcarusConfig) -> float:
        if redshift <= 0.0:
            return np.nan
        dl_mpc = config.globals.cosmology.luminosity_distance(redshift).value
        return float(dl_mpc) * MPC_TO_CM

    @staticmethod
    def _flux_to_luminosity(flux: float, dl_cm: float, scale_factor: float) -> float:
        if np.isnan(flux) or flux <= 0.0 or np.isnan(dl_cm):
            return np.nan
        flux_cgs = float(flux) * float(scale_factor)
        return float(flux_cgs * 4.0 * np.pi * (dl_cm**2))

    @staticmethod
    def _deconvolve_fwhm_kms(sigma_obs_kms: float, inst_sigma_kms: float) -> float:
        """Subtracts local instrumental velocity dispersion in quadrature."""
        if not np.isfinite(sigma_obs_kms) or sigma_obs_kms <= 0.0:
            return np.nan
        sigma_int = np.sqrt(
            max(float(sigma_obs_kms) ** 2 - float(inst_sigma_kms) ** 2, 0.0)
        )
        return float(sigma_int * FWHM_FACTOR)

    @staticmethod
    def _eval_monochromatic_lum(
        rest_wave_aa: float,
        wave_arr: np.ndarray,
        cont_eval: Optional[np.ndarray],
        dl_cm: float,
        scale_factor: float,
    ) -> float:
        """
        Computes monochromatic continuum luminosity lambda * L_lambda(rest_wave_aa) [erg/s]
        from the Stage 1 AGN continuum array.
        """
        if (
            cont_eval is None
            or len(cont_eval) != len(wave_arr)
            or np.isnan(dl_cm)
            or rest_wave_aa < np.min(wave_arr)
            or rest_wave_aa > np.max(wave_arr)
        ):
            return np.nan

        f_lambda_norm = float(np.interp(rest_wave_aa, wave_arr, cont_eval))
        if not np.isfinite(f_lambda_norm) or f_lambda_norm <= 0.0:
            return np.nan

        f_lambda_cgs = f_lambda_norm * float(scale_factor)
        return float(4.0 * np.pi * (dl_cm**2) * rest_wave_aa * f_lambda_cgs)

    @staticmethod
    def _resolve_line(
        clean: Dict[str, Any], raw: Dict[str, Any], aliases: List[str], suffix: str
    ) -> Tuple[float, bool]:
        for alias in aliases:
            key = f"{alias}_{suffix}"
            if key in clean and clean[key] != 0.0:
                is_ul = bool(raw.get(f"{alias}_narrow_is_upper_limit", False))
                return float(clean[key]), is_ul
        return 0.0, False

    @staticmethod
    def _check_kinematic_bound_hit(
        clean: Dict[str, Any], raw: Dict[str, Any], config: IcarusConfig
    ) -> bool:
        """
        Evaluates Bit 4 (16): Checks whether any detected narrow, broad, or outflow
        component pegged within 1% of its configured velocity dispersion (sigma_v)
        or velocity offset (v_offset) boundary wall.
        """
        fit_cfg = config.globals.fitting
        n_min_sig = (
            float(u.Quantity(fit_cfg.min_sigma).to(u.km / u.s).value)
            if fit_cfg.min_sigma is not None
            else 30.0
        )
        n_max_sig = (
            float(u.Quantity(fit_cfg.max_sigma).to(u.km / u.s).value)
            if fit_cfg.max_sigma is not None
            else 450.0
        )

        bk = fit_cfg.broad_kinematics or {}
        b_min_sig = (
            float(u.Quantity(bk["min_sigma"]).to(u.km / u.s).value)
            if bk.get("min_sigma") is not None
            else 800.0
        )
        b_max_sig = (
            float(u.Quantity(bk["max_sigma"]).to(u.km / u.s).value)
            if bk.get("max_sigma") is not None
            else 8000.0
        )

        def _near_wall(
            val: float, low: float, high: float, tol_frac: float = 0.01
        ) -> bool:
            span = max(abs(high - low), 1.0)
            return (val <= low + tol_frac * span) or (val >= high - tol_frac * span)

        for key, val in clean.items():
            if not isinstance(val, (int, float)) or not np.isfinite(val):
                continue

            # 1. Narrow components (only check non-upper-limit detections)
            if key.endswith("_narrow_sigma_v"):
                ion = key.removesuffix("_narrow_sigma_v")
                is_ul = bool(raw.get(f"{ion}_narrow_is_upper_limit", False))
                amp = float(clean.get(f"{ion}_narrow_amplitude", 0.0))
                if not is_ul and amp > 0.0:
                    if _near_wall(float(val), n_min_sig, n_max_sig):
                        return True
                    v_off = float(clean.get(f"{ion}_narrow_v_offset", 0.0))
                    if _near_wall(v_off, -350.0, 350.0):
                        return True

            # 2. Broad BLR components
            elif key.endswith("_broad_sigma_v"):
                ion = key.removesuffix("_broad_sigma_v")
                amp = float(clean.get(f"{ion}_broad_amplitude", 0.0))
                if amp > 0.0:
                    if _near_wall(float(val), b_min_sig, b_max_sig):
                        return True
                    v_off = float(clean.get(f"{ion}_broad_v_offset", 0.0))
                    is_uv = any(uv in ion for uv in UV_WIND_IONS)
                    v_low = -4000.0 if is_uv else -600.0
                    v_high = 1000.0 if is_uv else 600.0
                    if _near_wall(v_off, v_low, v_high):
                        return True

            # 3. Asymmetric Outflow components
            elif key.endswith("_outflow_sigma_v"):
                ion = key.removesuffix("_outflow_sigma_v")
                amp = float(clean.get(f"{ion}_outflow_amplitude", 0.0))
                if amp > 1e-4:
                    if _near_wall(float(val), 250.0, 765.0):
                        return True
                    v_off = float(clean.get(f"{ion}_outflow_v_offset", -350.0))
                    if _near_wall(v_off, -2000.0, -50.0):
                        return True

        return False

    @classmethod
    @dev_hook(step_name="compute_all_demographics")
    def compute_all(
        cls,
        row: Dict[str, Any],
        spectrum: Spectrum1D,
        config: IcarusConfig,
        global_cont_eval: Optional[np.ndarray] = None,
        feii_eval: Optional[np.ndarray] = None,
    ) -> Tuple[Dict[str, Any], int]:
        """
        Computes all derived AGN and host galaxy physical metrics and compiles
        the 6-bit additive `confidence_flag`:
          Bit 0 (1) : Local Continuum Fallback (Stage 1 global continuum unavailable)
          Bit 1 (2) : Fe II Template Active
          Bit 2 (4) : Broad BLR Component Confirmed
          Bit 3 (8) : [O III] Outflow Wing Confirmed
          Bit 4 (16): Parameter Hit Kinematic Boundary
          Bit 5 (32): All Lines Upper Limits (Zero 3-sigma detections)
        """
        DevState.emit("pre_astrophysics_calc", locals())

        target = spectrum.target
        scale_factor = spectrum.flux_scale_factor
        wave_arr, _, _ = spectrum.to_ndarrays()

        clean: Dict[str, Any] = {}
        for k, v in row.items():
            if k == "TargetID":
                clean[k] = str(v)
            else:
                try:
                    clean[k] = (
                        float(v) if v is not None and not np.isnan(float(v)) else 0.0
                    )
                except (ValueError, TypeError):
                    clean[k] = v

        metrics: Dict[str, Any] = {}
        dl_cm = cls._get_luminosity_distance_cm(target.redshift, config)

        # 1. Rest-Frame Equivalent Widths
        for key, val in clean.items():
            if key.endswith("_amplitude") and isinstance(val, (int, float)) and val > 0:
                line_prefix = key.replace("_amplitude", "")
                base_ion = (
                    line_prefix.replace("_narrow", "")
                    .replace("_broad", "")
                    .replace("_outflow", "")
                )
                cont_val = clean.get(
                    f"{base_ion}_continuum_flux",
                    clean.get("cont_intercept", clean.get("cont_c", 0.0)),
                )
                if isinstance(cont_val, (int, float)) and cont_val > 1e-6:
                    metrics[f"{line_prefix}_EW"] = float(val / cont_val)

        # 2. Monochromatic Continuum Luminosities (L_1350, L_3000, L_5100)
        l_1350 = cls._eval_monochromatic_lum(
            1350.0, wave_arr, global_cont_eval, dl_cm, scale_factor
        )
        l_3000 = cls._eval_monochromatic_lum(
            3000.0, wave_arr, global_cont_eval, dl_cm, scale_factor
        )
        l_5100 = cls._eval_monochromatic_lum(
            5100.0, wave_arr, global_cont_eval, dl_cm, scale_factor
        )
        metrics["L_1350_erg_s"] = l_1350
        metrics["L_3000_erg_s"] = l_3000
        metrics["L_5100_erg_s"] = l_5100

        # 3. Narrow Line Luminosities & BPT Diagnostics
        ha_n, ha_ul = cls._resolve_line(
            clean, row, ["H_alpha", "Ha"], "narrow_amplitude"
        )
        hb_n, hb_ul = cls._resolve_line(
            clean, row, ["H_beta", "Hb"], "narrow_amplitude"
        )
        pab_n, pab_ul = cls._resolve_line(
            clean, row, ["Pa_beta", "Pab"], "narrow_amplitude"
        )
        oiii_f, oiii_ul = cls._resolve_line(
            clean, row, ["O_III_5007"], "narrow_amplitude"
        )
        nii_f, nii_ul = cls._resolve_line(clean, row, ["N_II_6583"], "narrow_amplitude")
        neiii_f, neiii_ul = cls._resolve_line(
            clean, row, ["Ne_III_3869"], "narrow_amplitude"
        )
        oii_f, oii_ul = cls._resolve_line(
            clean, row, ["O_II_3729", "O_II_3726"], "narrow_amplitude"
        )

        if ha_n > 0:
            metrics["L_Ha_narrow_erg_s"] = cls._flux_to_luminosity(
                ha_n, dl_cm, scale_factor
            )
        if pab_n > 0:
            metrics["L_Pab_narrow_erg_s"] = cls._flux_to_luminosity(
                pab_n, dl_cm, scale_factor
            )

        # BPT Ratios + Upper-Limit Arrow Metadata for Panel 4
        if oiii_f > 0 and hb_n > 0 and not (oiii_ul and hb_ul):
            metrics["log_O3_Hb"] = float(np.log10(oiii_f / hb_n))
            metrics["O3_Hb_limit"] = "down" if oiii_ul else ("up" if hb_ul else "none")
        else:
            metrics["log_O3_Hb"] = np.nan
            metrics["O3_Hb_limit"] = "none"

        if nii_f > 0 and ha_n > 0 and not (nii_ul and ha_ul):
            metrics["log_N2_Ha"] = float(np.log10(nii_f / ha_n))
            metrics["N2_Ha_limit"] = (
                "left" if nii_ul else ("right" if ha_ul else "none")
            )
        else:
            metrics["log_N2_Ha"] = np.nan
            metrics["N2_Ha_limit"] = "none"

        if neiii_f > 0 and oii_f > 0 and not (neiii_ul and oii_ul):
            metrics["log_Ne3_O2"] = float(np.log10(neiii_f / oii_f))
        else:
            metrics["log_Ne3_O2"] = np.nan

        # Systemic narrow velocity anchor (for C IV blueshift calculation)
        v_sys_kms = 0.0
        for sys_ion in ("O_III_5007", "O_II_3729", "Ne_III_3869", "N_II_6583"):
            v_val, is_ul = cls._resolve_line(clean, row, [sys_ion], "narrow_v_offset")
            if not is_ul and f"{sys_ion}_narrow_v_offset" in row:
                v_sys_kms = v_val
                break

        # 4. Single-Epoch Virial Black Hole Mass Waterfall
        # Hierarchy: H_beta (VP06) -> H_alpha (GH05) -> Mg_II (VO09) -> C_IV (Coatman17) -> Pa_beta
        hb_b_f, _ = cls._resolve_line(clean, row, ["H_beta", "Hb"], "broad_amplitude")
        hb_b_sv, _ = cls._resolve_line(clean, row, ["H_beta", "Hb"], "broad_sigma_v")

        ha_b_f, _ = cls._resolve_line(clean, row, ["H_alpha", "Ha"], "broad_amplitude")
        ha_b_sv, _ = cls._resolve_line(clean, row, ["H_alpha", "Ha"], "broad_sigma_v")

        mg_b_f, _ = cls._resolve_line(clean, row, ["Mg_II_2796"], "broad_amplitude")
        mg_b_sv, _ = cls._resolve_line(clean, row, ["Mg_II_2796"], "broad_sigma_v")

        civ_b_f, _ = cls._resolve_line(clean, row, ["C_IV_1548"], "broad_amplitude")
        civ_b_sv, _ = cls._resolve_line(clean, row, ["C_IV_1548"], "broad_sigma_v")
        civ_b_voff, _ = cls._resolve_line(clean, row, ["C_IV_1548"], "broad_v_offset")

        pab_b_f, _ = cls._resolve_line(
            clean, row, ["Pa_beta", "Pab"], "broad_amplitude"
        )
        pab_b_sv, _ = cls._resolve_line(clean, row, ["Pa_beta", "Pab"], "broad_sigma_v")

        mbh_msun = np.nan
        lbol_erg_s = np.nan
        fwhm_broad_kms = np.nan
        mbh_cal = "None"

        if hb_b_f > 0 and hb_b_sv > 0:
            l_hb_b = cls._flux_to_luminosity(hb_b_f, dl_cm, scale_factor)
            metrics["L_Hb_broad_erg_s"] = l_hb_b
            fwhm_hb = cls._deconvolve_fwhm_kms(
                hb_b_sv, spectrum.get_inst_sigma_kms(4862.68)
            )
            fwhm_broad_kms = fwhm_hb
            l5100_eff = (
                l_5100 if (np.isfinite(l_5100) and l_5100 > 0) else (l_hb_b * 113.0)
            )
            lbol_erg_s = l5100_eff * 9.26
            mbh_msun = (
                (10.0**6.91)
                * ((fwhm_hb / 1000.0) ** 2.0)
                * ((l5100_eff / 1e44) ** 0.50)
            )
            mbh_cal = "Hbeta_VP06"

        elif ha_b_f > 0 and ha_b_sv > 0:
            l_ha_b = cls._flux_to_luminosity(ha_b_f, dl_cm, scale_factor)
            metrics["L_Ha_broad_erg_s"] = l_ha_b
            fwhm_ha = cls._deconvolve_fwhm_kms(
                ha_b_sv, spectrum.get_inst_sigma_kms(6564.61)
            )
            fwhm_broad_kms = fwhm_ha
            lbol_erg_s = (
                l_5100 * 9.26
                if (np.isfinite(l_5100) and l_5100 > 0)
                else l_ha_b * 130.0
            )
            mbh_msun = 2.0e6 * ((l_ha_b / 1e42) ** 0.55) * ((fwhm_ha / 1000.0) ** 2.06)
            mbh_cal = "Halpha_GH05"

        elif mg_b_f > 0 and mg_b_sv > 0:
            l_mg_b = cls._flux_to_luminosity(mg_b_f, dl_cm, scale_factor)
            fwhm_mg = cls._deconvolve_fwhm_kms(
                mg_b_sv, spectrum.get_inst_sigma_kms(2796.35)
            )
            fwhm_broad_kms = fwhm_mg
            l3000_eff = (
                l_3000 if (np.isfinite(l_3000) and l_3000 > 0) else (l_mg_b * 55.0)
            )
            lbol_erg_s = l3000_eff * 5.15
            mbh_msun = (
                (10.0**6.86)
                * ((fwhm_mg / 1000.0) ** 2.0)
                * ((l3000_eff / 1e44) ** 0.50)
            )
            mbh_cal = "MgII_VO09"

        elif civ_b_f > 0 and civ_b_sv > 0:
            l_civ_b = cls._flux_to_luminosity(civ_b_f, dl_cm, scale_factor)
            fwhm_civ_meas = cls._deconvolve_fwhm_kms(
                civ_b_sv, spectrum.get_inst_sigma_kms(1548.19)
            )
            # Coatman et al. (2017) non-virial blueshift correction
            dv_blue_kms = max(float(v_sys_kms - civ_b_voff), 0.0)
            metrics["CIV_blueshift_kms"] = dv_blue_kms
            if dv_blue_kms > 0.0:
                fwhm_civ_corr = fwhm_civ_meas / (0.36 * (dv_blue_kms / 1000.0) + 0.61)
            else:
                fwhm_civ_corr = fwhm_civ_meas

            fwhm_broad_kms = fwhm_civ_corr
            l1350_eff = (
                l_1350 if (np.isfinite(l_1350) and l_1350 > 0) else (l_civ_b * 35.0)
            )
            lbol_erg_s = l1350_eff * 3.81
            mbh_msun = (
                (10.0**6.66)
                * ((fwhm_civ_corr / 1000.0) ** 2.0)
                * ((l1350_eff / 1e44) ** 0.53)
            )
            mbh_cal = "CIV_Coatman17"

        elif pab_b_f > 0 and pab_b_sv > 0:
            l_pab_b = cls._flux_to_luminosity(pab_b_f, dl_cm, scale_factor)
            metrics["L_Pab_broad_erg_s"] = l_pab_b
            l_ha_equiv = l_pab_b * 17.5
            fwhm_pab = cls._deconvolve_fwhm_kms(
                pab_b_sv, spectrum.get_inst_sigma_kms(12821.59)
            )
            fwhm_broad_kms = fwhm_pab
            lbol_erg_s = l_ha_equiv * 130.0
            mbh_msun = (
                2.0e6 * ((l_ha_equiv / 1e42) ** 0.55) * ((fwhm_pab / 1000.0) ** 2.06)
            )
            mbh_cal = "PaBeta_CaseB"

        metrics["M_BH_Msun"] = mbh_msun
        metrics["M_BH_Calibrator"] = mbh_cal
        metrics["M_BH_sys_err_dex"] = 0.35 if np.isfinite(mbh_msun) else np.nan
        metrics["L_bol_erg_s"] = lbol_erg_s
        metrics["FWHM_broad_kms"] = fwhm_broad_kms

        if np.isfinite(mbh_msun) and mbh_msun > 0 and np.isfinite(lbol_erg_s):
            l_edd = 1.26e38 * mbh_msun
            metrics["lambda_Edd"] = float(lbol_erg_s / l_edd)
        else:
            metrics["lambda_Edd"] = np.nan

        # 5. Optical Fe II Relative Strength (R_FeII = F_FeII(4434-4684) / F_Hb_broad)
        if (
            feii_eval is not None
            and len(feii_eval) == len(wave_arr)
            and hb_b_f > 0
            and _TRAPZ is not None
        ):
            opt_fe_mask = (wave_arr >= 4434.0) & (wave_arr <= 4684.0) & (~spectrum.mask)
            if np.sum(opt_fe_mask) >= 3:
                fe_int_flux = float(
                    _TRAPZ(feii_eval[opt_fe_mask], wave_arr[opt_fe_mask])
                )
                metrics["R_FeII"] = (
                    float(fe_int_flux / hb_b_f) if fe_int_flux > 0 else 0.0
                )
            else:
                metrics["R_FeII"] = np.nan
        else:
            metrics["R_FeII"] = np.nan

        # 6. [O III] Outflow Velocity Readout
        oiii_out_amp, _ = cls._resolve_line(
            clean, row, ["O_III_5007"], "outflow_amplitude"
        )
        oiii_out_voff, _ = cls._resolve_line(
            clean, row, ["O_III_5007"], "outflow_v_offset"
        )
        metrics["v_outflow_OIII_kms"] = (
            float(oiii_out_voff) if oiii_out_amp > 1e-4 else np.nan
        )

        # 7. Narrow-Line Star Formation Rate (Kennicutt & Evans 2012)
        sfr_ha = 5.37e-42 * metrics.get("L_Ha_narrow_erg_s", 0.0)
        sfr_pab = (10.0**-40.02) * metrics.get("L_Pab_narrow_erg_s", 0.0)

        if ha_n > 0 and not ha_ul and ha_b_f == 0:
            metrics["SFR_uncorrected_Msun_yr"] = float(sfr_ha)
            metrics["SFR_Indicator"] = "Halpha"
            metrics["SFR_is_upper_limit"] = False
        elif pab_n > 0 and not pab_ul and pab_b_f == 0:
            metrics["SFR_uncorrected_Msun_yr"] = float(sfr_pab)
            metrics["SFR_Indicator"] = "PaBeta"
            metrics["SFR_is_upper_limit"] = False
        elif ha_n > 0:
            metrics["SFR_uncorrected_Msun_yr"] = float(sfr_ha)
            metrics["SFR_Indicator"] = "Halpha"
            metrics["SFR_is_upper_limit"] = True if ha_b_f > 0 else ha_ul
        elif pab_n > 0:
            metrics["SFR_uncorrected_Msun_yr"] = float(sfr_pab)
            metrics["SFR_Indicator"] = "PaBeta"
            metrics["SFR_is_upper_limit"] = True if pab_b_f > 0 else pab_ul
        else:
            metrics["SFR_uncorrected_Msun_yr"] = np.nan
            metrics["SFR_Indicator"] = "None"
            metrics["SFR_is_upper_limit"] = False

        # 8. Additive Confidence Bitmask Evaluation
        conf_flag = 0
        if global_cont_eval is None:
            conf_flag |= 1  # Bit 0: Local continuum fallback
        if feii_eval is not None and np.nanmax(feii_eval) > 0:
            conf_flag |= 2  # Bit 1: Fe II template active
        if np.isfinite(mbh_msun) and mbh_msun > 0:
            conf_flag |= 4  # Bit 2: Broad BLR component confirmed
        if oiii_out_amp > 1e-4:
            conf_flag |= 8  # Bit 3: [O III] outflow wing confirmed
        if cls._check_kinematic_bound_hit(clean, row, config):
            conf_flag |= 16  # Bit 4: Detected component pegged against kinematic bound

        ul_flags = [
            bool(v) for k, v in row.items() if k.endswith("_narrow_is_upper_limit")
        ]
        if ul_flags and all(ul_flags):
            conf_flag |= 32  # Bit 5: All lines are 3-sigma upper limits

        DevState.emit("post_astrophysics_calc", locals())
        return metrics, conf_flag
"""
feii_model.py

Empirical UV and Optical Fe II pseudo-continuum FFT convolution engine for ICARUS v3.4.
Convolves Vestergaard & Wilkes (2001) / Tsuzuki et al. (2006) UV and Boroson & Green (1992)
I Zw 1 optical templates on a uniform logarithmic wavelength grid (dv = 10.0 km/s)
with intrinsic template width subtraction.
"""

__project__ = "ICARUS"
__author__ = "Joseph Havens"
__version__ = "3.4.0"

from pathlib import Path
from typing import Optional, Tuple, Union

import numpy as np
from astropy import constants as const
from astropy.table import QTable
from lmfit import Model
from scipy.signal import fftconvolve

from ..core.containers import SpectralComponent
from ..core.dev_tools import DevState, dev_hook
from ..core.logger import get_icarus_logger

log = get_icarus_logger()
C_KMS = const.c.to("km/s").value
INTRINSIC_IZW1_SIGMA_KMS = 900.0 / (2.0 * np.sqrt(2.0 * np.log(2.0)))  # 382.19 km/s
DV_GRID_KMS = 10.0
PACKAGE_DATA_DIR = Path(__file__).resolve().parent.parent / "data" / "templates"


class FeIITemplateConvolver:
    """
    Manages uniform log-wavelength grids and fast FFT Gaussian broadening
    for UV (1200-3500 A) and Optical (3500-7500 A) Fe II pseudo-continua.
    """

    def __init__(
        self,
        uv_template_path: Optional[Union[str, Path]] = None,
        opt_template_path: Optional[Union[str, Path]] = None,
    ) -> None:
        self.uv_path = (
            Path(uv_template_path)
            if uv_template_path
            else PACKAGE_DATA_DIR / "feii_uv_vw01_tz06.fits"
        )
        self.opt_path = (
            Path(opt_template_path)
            if opt_template_path
            else PACKAGE_DATA_DIR / "feii_opt_bg92.fits"
        )

        self.ln_wave_grid, self.wave_grid, self.uv_flux, self.opt_flux = (
            self._load_or_synthesize_templates()
        )

    def _load_or_synthesize_templates(
        self,
    ) -> Tuple[np.ndarray, np.ndarray, np.ndarray, np.ndarray]:
        """
        Loads the log-wavelength UV and Optical Fe II templates from disk,
        or synthesizes an analytic multiplet basis at intrinsic I Zw 1 resolution
        (FWHM = 900 km/s) if the FITS templates have not been generated yet.
        """
        d_ln_lam = DV_GRID_KMS / C_KMS
        ln_min = np.log(1000.0)
        ln_max = np.log(10000.0)
        ln_grid = np.arange(ln_min, ln_max, d_ln_lam, dtype=np.float64)
        wave_grid = np.exp(ln_grid)

        uv_arr = np.zeros_like(wave_grid)
        opt_arr = np.zeros_like(wave_grid)

        loaded_uv = False
        if self.uv_path.exists():
            try:
                tbl_uv = QTable.read(self.uv_path)
                w_u = np.asarray(tbl_uv[tbl_uv.colnames[0]], dtype=np.float64)
                f_u = np.asarray(tbl_uv[tbl_uv.colnames[1]], dtype=np.float64)
                uv_arr = np.interp(wave_grid, w_u, f_u, left=0.0, right=0.0)
                loaded_uv = True
            except Exception as e:
                log.warning(
                    f"Failed to read UV Fe II template ({self.uv_path.name}): {e}"
                )

        loaded_opt = False
        if self.opt_path.exists():
            try:
                tbl_opt = QTable.read(self.opt_path)
                w_o = np.asarray(tbl_opt[tbl_opt.colnames[0]], dtype=np.float64)
                f_o = np.asarray(tbl_opt[tbl_opt.colnames[1]], dtype=np.float64)
                opt_arr = np.interp(wave_grid, w_o, f_o, left=0.0, right=0.0)
                loaded_opt = True
            except Exception as e:
                log.warning(
                    f"Failed to read Optical Fe II template ({self.opt_path.name}): {e}"
                )

        if not loaded_uv:
            # Analytic UV Fe II multiplet complexes (UV 1, 2, 3, 34, 60, 62, 63, 78, Balmer wings)
            uv_multiplets = [
                (1610.0, 45.0, 0.35),
                (1715.0, 40.0, 0.30),
                (1860.0, 55.0, 0.45),
                (2100.0, 50.0, 0.35),
                (2260.0, 45.0, 0.50),
                (2340.0, 55.0, 0.85),
                (2385.0, 45.0, 0.95),
                (2445.0, 50.0, 0.90),
                (2505.0, 55.0, 1.00),
                (2565.0, 45.0, 0.85),
                (2615.0, 40.0, 0.90),
                (2745.0, 35.0, 0.75),
                (2850.0, 40.0, 0.55),
                (2950.0, 55.0, 0.65),
                (3210.0, 65.0, 0.45),
            ]
            for cen, width_aa, rel_amp in uv_multiplets:
                sig_tot = np.hypot(width_aa, cen * (INTRINSIC_IZW1_SIGMA_KMS / C_KMS))
                uv_arr += rel_amp * np.exp(-0.5 * ((wave_grid - cen) / sig_tot) ** 2)

        if not loaded_opt:
            # Analytic Optical Fe II multiplet complexes (m37, m38, m42, m48, m49)
            opt_multiplets = [
                # Blue shelf (4434 - 4684 A, m37 & m38)
                (4491.4, 0.45),
                (4508.3, 0.55),
                (4515.3, 0.65),
                (4520.2, 0.60),
                (4522.6, 0.70),
                (4549.5, 0.95),
                (4555.9, 0.85),
                (4576.3, 0.50),
                (4583.8, 1.00),
                (4620.5, 0.40),
                (4629.3, 0.75),
                (4666.8, 0.35),
                # m42 multiplet (4924, 5018, 5169 A)
                (4923.9, 0.65),
                (5018.4, 0.75),
                (5169.0, 0.85),
                # Red shelf (5150 - 5400 A, m48 & m49)
                (5197.6, 0.70),
                (5234.6, 0.80),
                (5264.8, 0.45),
                (5276.0, 0.85),
                (5284.1, 0.40),
                (5316.6, 0.95),
                (5325.6, 0.45),
                (5362.9, 0.65),
            ]
            for cen, rel_amp in opt_multiplets:
                sig_aa = cen * (INTRINSIC_IZW1_SIGMA_KMS / C_KMS)
                opt_arr += rel_amp * np.exp(-0.5 * ((wave_grid - cen) / sig_aa) ** 2)

        # Normalize peak of each template to 1.0 so amplitude parameters represent peak flux density
        if np.max(uv_arr) > 0:
            uv_arr = uv_arr / np.max(uv_arr)
        if np.max(opt_arr) > 0:
            opt_arr = opt_arr / np.max(opt_arr)

        return ln_grid, wave_grid, uv_arr, opt_arr

    @dev_hook(step_name="convolve_feii_template")
    def convolve_and_project(
        self,
        x: np.ndarray,
        feii_uv_amp: float = 0.0,
        feii_opt_amp: float = 0.0,
        feii_v_offset: float = 0.0,
        feii_sigma_v: float = 1200.0,
    ) -> np.ndarray:
        """
        Broadens the combined UV + Optical Fe II template on the uniform log-wavelength
        grid via FFT Gaussian convolution (accounting for the 382.19 km/s intrinsic
        I Zw 1 width) and projects onto the target rest-frame wavelength array `x`.
        """
        if feii_uv_amp <= 1e-12 and feii_opt_amp <= 1e-12:
            return np.zeros_like(x, dtype=np.float64)

        base_template = (
            max(float(feii_uv_amp), 0.0) * self.uv_flux
            + max(float(feii_opt_amp), 0.0) * self.opt_flux
        )

        # Subtract intrinsic I Zw 1 dispersion in quadrature
        clamped_sigma = max(float(feii_sigma_v), 400.0)
        sigma_kern_kms = np.sqrt(
            max(clamped_sigma**2 - INTRINSIC_IZW1_SIGMA_KMS**2, DV_GRID_KMS**2)
        )
        sigma_pix = sigma_kern_kms / DV_GRID_KMS

        half_width = max(int(np.ceil(5.0 * sigma_pix)), 3)
        kx = np.arange(-half_width, half_width + 1, dtype=np.float64)
        kernel = np.exp(-0.5 * (kx / sigma_pix) ** 2)
        kernel /= np.sum(kernel)

        broadened = fftconvolve(base_template, kernel, mode="same")

        # Apply velocity shift in log-wavelength space: ln(lambda_shifted) = ln(lambda) + v / c
        shifted_wave = self.wave_grid * (1.0 + float(feii_v_offset) / C_KMS)
        projected = np.interp(x, shifted_wave, broadened, left=0.0, right=0.0)
        return np.asarray(projected, dtype=np.float64)

    def build_lmfit_model(
        self,
        prefix: str = "",
        uv_amp_guess: float = 0.05,
        opt_amp_guess: float = 0.05,
        max_amp: float = 100.0,
    ) -> SpectralComponent:
        """Wraps `convolve_and_project` as a constrainable LMFIT Model."""

        def _feii_func(
            x: np.ndarray,
            feii_uv_amp: float = 0.05,
            feii_opt_amp: float = 0.05,
            feii_v_offset: float = 0.0,
            feii_sigma_v: float = 1200.0,
        ) -> np.ndarray:
            return self.convolve_and_project(
                x,
                feii_uv_amp=feii_uv_amp,
                feii_opt_amp=feii_opt_amp,
                feii_v_offset=feii_v_offset,
                feii_sigma_v=feii_sigma_v,
            )

        model = Model(_feii_func, prefix=prefix)
        params = model.make_params()

        safe_max = max(float(max_amp), 1.0)
        params[f"{prefix}feii_uv_amp"].set(
            value=max(float(uv_amp_guess), 0.0), min=0.0, max=safe_max
        )
        params[f"{prefix}feii_opt_amp"].set(
            value=max(float(opt_amp_guess), 0.0), min=0.0, max=safe_max
        )
        params[f"{prefix}feii_v_offset"].set(value=0.0, min=-500.0, max=500.0)
        params[f"{prefix}feii_sigma_v"].set(value=1200.0, min=400.0, max=5000.0)

        return SpectralComponent(
            name=prefix.strip("_") or "feii",
            model=model,
            params=params,
            component_type="feii_template",
        )

class SpectralPreparer:
    """Handles dynamic wavelength proximity grouping and boundary-safe complex slicing."""

    @staticmethod
    @dev_hook(step_name="group_into_complexes")
    def group_into_complexes(
        lines: List[EmissionLineDef],
        config: FittingConfig,
        narrow_merge_radius: float = 25.0,
        broad_merge_radius: float = 120.0,
    ) -> List[LineComplex]:
        if not lines:
            return []

        sorted_lines = sorted(lines, key=lambda x: x.rest_wavelength)
        complexes: List[LineComplex] = []
        current_group = [sorted_lines[0]]

        current_end = sorted_lines[0].rest_wavelength + (
            broad_merge_radius if sorted_lines[0].test_broad else narrow_merge_radius
        )

        pad = (
            (config.cont_width.to(u.AA).value / 2.0)
            if config.cont_width is not None
            else 50.0
        )

        for line in sorted_lines[1:]:
            line_start = line.rest_wavelength - (
                broad_merge_radius if line.test_broad else narrow_merge_radius
            )
            line_end = line.rest_wavelength + (
                broad_merge_radius if line.test_broad else narrow_merge_radius
            )

            if line_start <= current_end:
                current_group.append(line)
                current_end = max(current_end, line_end)
            else:
                complexes.append(
                    LineComplex(
                        lines=current_group,
                        window_start=(current_group[0].rest_wavelength - pad) * u.AA,
                        window_end=(current_group[-1].rest_wavelength + pad) * u.AA,
                    )
                )
                current_group = [line]
                current_end = line_end

        complexes.append(
            LineComplex(
                lines=current_group,
                window_start=(current_group[0].rest_wavelength - pad) * u.AA,
                window_end=(current_group[-1].rest_wavelength + pad) * u.AA,
            )
        )
        return complexes

    @staticmethod
    @dev_hook(step_name="slice_spectrum")
    def slice_spectrum(spectrum: Spectrum1D, comp: LineComplex) -> Spectrum1D:
        start = comp.window_start.to(spectrum.wavelength.unit).value
        end = comp.window_end.to(spectrum.wavelength.unit).value

        wave_val = spectrum.wavelength.value
        valid_mask = (
            (~spectrum.mask) & np.isfinite(wave_val) & np.isfinite(spectrum.flux.value)
        )
        window_mask = (wave_val >= start) & (wave_val <= end) & valid_mask

        if np.sum(window_mask) < 8:
            raise IcarusDataDimensionError(
                f"Complex {comp.name} falls outside valid data range or has insufficient unmasked pixels."
            )

        sliced_wave = wave_val[window_mask]

        # Prune individual lines whose core (+/- 12 A) sits on NaNs or off the detector
        active_lines = [
            line
            for line in comp.lines
            if np.sum(np.abs(sliced_wave - line.rest_wavelength) <= 12.0) >= 2
        ]

        if not active_lines:
            raise IcarusDataDimensionError(
                f"All emission line cores in {comp.name} fall on masked/NaN regions or off detector."
            )

        comp.lines = active_lines
        min_line_w = min(l.rest_wavelength for l in active_lines)
        max_line_w = max(l.rest_wavelength for l in active_lines)

        has_left_cont = np.min(sliced_wave) <= (min_line_w - 30.0)
        has_right_cont = np.max(sliced_wave) >= (max_line_w + 30.0)
        span_ratio = (np.max(sliced_wave) - np.min(sliced_wave)) / max(end - start, 1.0)
        comp.in_bounds = bool(has_left_cont and has_right_cont and (span_ratio >= 0.65))

        sliced_res = (
            spectrum.resolution_kms[window_mask]
            if spectrum.resolution_kms is not None
            else None
        )

        DevState.emit("post_slice_spectrum", locals())

        return Spectrum1D(
            wavelength=spectrum.wavelength[window_mask],
            flux=spectrum.flux[window_mask],
            uncertainty=spectrum.uncertainty[window_mask],
            mask=np.zeros(np.sum(window_mask), dtype=bool),
            target=spectrum.target,
            spectral_unit=spectrum.spectral_unit,
            flux_unit=spectrum.flux_unit,
            flux_scale_factor=spectrum.flux_scale_factor,
            resolution_kms=sliced_res,
            meta=spectrum.meta,
            history=spectrum.history + [f"Sliced to complex {comp.name}"],
        )

    @staticmethod
    @dev_hook(step_name="estimate_initial_amplitude")
    def estimate_initial_amplitude(
        sliced_spec: Spectrum1D,
        rest_wavelength: float,
        search_window_aa: float = 10.0,
        cont_eval: Optional[np.ndarray] = None,
    ) -> float:
        wave_vals, flux_vals, _ = sliced_spec.to_ndarrays()

        if cont_eval is not None and len(cont_eval) == len(flux_vals):
            sub_flux = flux_vals - cont_eval
        else:
            _, median_cont, _ = sigma_clipped_stats(flux_vals, sigma=2.5, maxiters=5)
            sub_flux = flux_vals - median_cont

        core_mask = (wave_vals >= rest_wavelength - search_window_aa) & (
            wave_vals <= rest_wavelength + search_window_aa
        )

        if not np.any(core_mask):
            return 1e-15

        peak_height = max(float(np.nanmax(sub_flux[core_mask])), 1e-5)
        inst_sig_kms = sliced_spec.get_inst_sigma_kms(rest_wavelength)
        rough_sigma_guess = max(rest_wavelength * (inst_sig_kms / 299792.458), 1.5)
        amplitude_guess = peak_height * rough_sigma_guess * SQRT_2_PI

        DevState.emit("post_amplitude_estimation", locals())
        return float(amplitude_guess)


class CoreFitter:
    """Builds constrained complex models and executes BIC broad-line selection."""

    @staticmethod
    @dev_hook(step_name="build_complex_model")
    def build_complex_model(
        comp: LineComplex,
        sliced_spec: Spectrum1D,
        config: FittingConfig,
        include_broad: bool = False,
        continuum: Any = None,
        **kwargs: Any,
    ) -> Tuple[Any, Any]:
        x_vals, _, _ = sliced_spec.to_ndarrays()

        # 1. Stage 2 Local Boundary-Condition Continuum Pinning
        cont_comp = ContinuumFitter.build_local_boundary_continuum(
            sliced_spec=sliced_spec, lines=comp.lines, prefix="cont_"
        )
        model = cont_comp.model
        params = cont_comp.params.copy()
        cont_init_eval = model.eval(params, x=x_vals)

        # 2. Iteratively construct narrow, broad, and outflow components
        built_lines: List[EmissionLineDef] = []
        for line in comp.lines:
            if len(sliced_spec.wavelength) < len(model.param_names) + 4:
                continue

            amp_guess = SpectralPreparer.estimate_initial_amplitude(
                sliced_spec, line.rest_wavelength, cont_eval=cont_init_eval
            )
            inst_sig = sliced_spec.get_inst_sigma_kms(line.rest_wavelength)

            n_comp = LineComponentFactory.build_emission_line(
                name=f"{line.name}_narrow",
                rest_wavelength=line.rest_wavelength,
                is_broad=False,
                amplitude_guess=amp_guess,
                config=config,
                sigma_init_kms=max(inst_sig, 80.0),
            )
            model += n_comp.model
            params.update(n_comp.params)
            built_lines.append(line)

            if include_broad and line.test_broad:
                b_comp = LineComponentFactory.build_emission_line(
                    name=f"{line.name}_broad",
                    rest_wavelength=line.rest_wavelength,
                    is_broad=True,
                    amplitude_guess=amp_guess * 0.5,
                    config=config,
                )
                model += b_comp.model
                params.update(b_comp.params)

            if line.outflow_candidate or line.name in ("O_III_4959", "O_III_5007"):
                outflow_comp = LineComponentFactory.build_outflow_line(
                    name=f"{line.name}_outflow",
                    rest_wavelength=line.rest_wavelength,
                    amplitude_guess=amp_guess * 0.2,
                )
                model += outflow_comp.model
                params.update(outflow_comp.params)

        # 3. Systemic Narrow Kinematic Coupling within a Multi-Line Complex
        if getattr(config, "couple_narrow_v_offset", True) and len(built_lines) > 1:
            anchor_line = comp.primary_line
            anchor_v = f"{anchor_line.name}_narrow_v_offset"
            for line in built_lines:
                if line.name != anchor_line.name:
                    v_key = f"{line.name}_narrow_v_offset"
                    if v_key in params and anchor_v in params:
                        params[v_key].set(expr=anchor_v)

        # 4. Quantum Mechanical Doublet Tying Matrix
        if config.doublet_tying:
            for _, definition in config.doublet_tying.items():
                target_lines = definition.get("lines", [])
                if len(target_lines) != 2:
                    continue

                line_left, line_right = target_lines[0], target_lines[1]
                l_amp = f"{line_left}_narrow_amplitude"
                r_amp = f"{line_right}_narrow_amplitude"

                if l_amp in params and r_amp in params:
                    flux_ratio = definition.get("flux_ratio")
                    if flux_ratio is not None:
                        params[l_amp].set(expr=f"{r_amp} * {flux_ratio}")

                    if definition.get("tie_kinematics", False):
                        params[f"{line_left}_narrow_sigma_v"].set(
                            expr=f"{line_right}_narrow_sigma_v"
                        )
                        params[f"{line_left}_narrow_v_offset"].set(
                            expr=f"{line_right}_narrow_v_offset"
                        )

                    l_b_amp = f"{line_left}_broad_amplitude"
                    r_b_amp = f"{line_right}_broad_amplitude"
                    if include_broad and l_b_amp in params and r_b_amp in params:
                        if flux_ratio is not None:
                            params[l_b_amp].set(expr=f"{r_b_amp} * {flux_ratio}")
                        params[f"{line_left}_broad_sigma_v"].set(
                            expr=f"{line_right}_broad_sigma_v"
                        )
                        params[f"{line_left}_broad_v_offset"].set(
                            expr=f"{line_right}_broad_v_offset"
                        )

                    l_out_amp = f"{line_left}_outflow_amplitude"
                    r_out_amp = f"{line_right}_outflow_amplitude"
                    if l_out_amp in params and r_out_amp in params:
                        if flux_ratio is not None:
                            params[l_out_amp].set(expr=f"{r_out_amp} * {flux_ratio}")
                        params[f"{line_left}_outflow_sigma_v"].set(
                            expr=f"{line_right}_outflow_sigma_v"
                        )
                        params[f"{line_left}_outflow_v_offset"].set(
                            expr=f"{line_right}_outflow_v_offset"
                        )

        DevState.emit("post_model_build", locals())
        return model, params

    @staticmethod
    @dev_hook(step_name="evaluate_broad_line")
    def evaluate_broad_line(
        comp: LineComplex,
        sliced_spec: Spectrum1D,
        config: FittingConfig,
        continuum: Any = None,
        feii_eval: Optional[np.ndarray] = None,
        full_wave: Optional[np.ndarray] = None,
        **kwargs: Any,
    ) -> ModelResult:
        x_data, y_raw, err_data = sliced_spec.to_ndarrays()
        weights = 1.0 / np.maximum(err_data, 1e-5)

        # Subtract Stage 1 Fe II pseudo-continuum if active over this complex window
        fe_slice = np.zeros_like(y_raw)
        if feii_eval is not None and np.nanmax(feii_eval) > 0:
            if len(feii_eval) == len(y_raw):
                fe_slice = np.asarray(feii_eval, dtype=np.float64)
            elif full_wave is not None and len(full_wave) == len(feii_eval):
                fe_slice = np.interp(x_data, full_wave, feii_eval, left=0.0, right=0.0)

        y_data = y_raw - fe_slice
        fit_spec = (
            Spectrum1D(
                wavelength=sliced_spec.wavelength,
                flux=y_data * sliced_spec.flux_unit,
                uncertainty=sliced_spec.uncertainty,
                mask=sliced_spec.mask,
                target=sliced_spec.target,
                spectral_unit=sliced_spec.spectral_unit,
                flux_unit=sliced_spec.flux_unit,
                flux_scale_factor=sliced_spec.flux_scale_factor,
                resolution_kms=sliced_spec.resolution_kms,
                meta=sliced_spec.meta,
                history=sliced_spec.history,
            )
            if np.any(fe_slice > 0)
            else sliced_spec
        )

        mod_a, pars_a = CoreFitter.build_complex_model(
            comp, fit_spec, config, include_broad=False, continuum=continuum
        )

        DevState.emit("pre_narrow_fit", locals())
        result_a = mod_a.fit(y_data, pars_a, x=x_data, weights=weights)
        DevState.emit("post_narrow_fit", locals())

        if not any(line.test_broad for line in comp.lines):
            if not result_a.success:
                raise IcarusConvergenceError(
                    f"Narrow-only fit failed to converge for complex: {comp.name}"
                )
            return result_a

        res_a = y_data - result_a.best_fit
        local_rms = max(
            float(np.percentile(np.abs(res_a - np.median(res_a)), 68)), 1e-5
        )

        # Speed Gatekeeper: Skip broad optimization pass if narrow core has < 2.0x RMS signal
        if getattr(config, "enable_speed_gatekeeper", True):
            has_core_signal = False
            for line in comp.lines:
                if line.test_broad:
                    n_amp = result_a.params.get(f"{line.name}_narrow_amplitude")
                    n_sig = result_a.params.get(f"{line.name}_narrow_sigma")
                    if n_amp and n_sig and n_sig.value > 0:
                        if (n_amp.value / (n_sig.value * SQRT_2_PI)) >= 2.0 * local_rms:
                            has_core_signal = True
                            break

            if not has_core_signal:
                return result_a

        mod_b, pars_b = CoreFitter.build_complex_model(
            comp, fit_spec, config, include_broad=True, continuum=continuum
        )

        DevState.emit("pre_broad_fit", locals())
        result_b = mod_b.fit(y_data, pars_b, x=x_data, weights=weights)
        DevState.emit("post_broad_fit", locals())

        if not result_a.success and not result_b.success:
            raise IcarusConvergenceError(
                f"Both structural models failed to converge for complex: {comp.name}"
            )
        if not result_b.success:
            return result_a
        if not result_a.success:
            return result_b

        DevState.emit("pre_bic_evaluation", locals())

        # Physical Broad Veto Checks
        window_span = float(np.max(x_data) - np.min(x_data))
        any_valid_broad = False
        veto_broad = False

        for line in comp.lines:
            if not line.test_broad:
                continue

            b_amp = result_b.params.get(f"{line.name}_broad_amplitude")
            n_amp = result_b.params.get(f"{line.name}_narrow_amplitude")
            b_sig = result_b.params.get(f"{line.name}_broad_sigma")
            n_sig = result_b.params.get(f"{line.name}_narrow_sigma")

            if (
                b_amp
                and n_amp
                and b_sig
                and n_sig
                and b_sig.value > 0
                and n_sig.value > 0
            ):
                b_peak = b_amp.value / (b_sig.value * SQRT_2_PI)
                n_peak = n_amp.value / (n_sig.value * SQRT_2_PI)
                b_fwhm_aa = b_sig.value * 2.3548

                # 1. Continuum-soak veto: broad FWHM > 70% of window span
                if b_fwhm_aa > 0.70 * window_span:
                    veto_broad = True
                    break

                # 2. Amplitude floor veto: broad peak < 5% of narrow peak
                if n_peak > 0 and b_peak < 0.05 * n_peak:
                    veto_broad = True
                    break

                # 3. Minimum SNR veto: broad peak must clear 2.5x local RMS
                if b_peak >= 2.5 * local_rms:
                    any_valid_broad = True

        if veto_broad or not any_valid_broad:
            DevState.emit("post_bic_evaluation", locals())
            return result_a

        delta_bic = result_b.bic - result_a.bic
        if delta_bic <= -10.0:
            log.info(
                f"Broad component confirmed for {comp.name} (Delta BIC: {delta_bic:.1f})"
            )
            DevState.emit("post_bic_evaluation", locals())
            return result_b

        DevState.emit("post_bic_evaluation", locals())
        return result_a

    @staticmethod
    @dev_hook(step_name="estimate_errors_boxcar")
    def estimate_errors_boxcar(
        sliced_spec: Spectrum1D, winning_result: ModelResult, boxcar_width: int = 5
    ) -> Dict[str, Tuple[float, float, float]]:
        _, base_flux, err_flux = sliced_spec.to_ndarrays()

        kernel = np.ones(boxcar_width) / float(boxcar_width)
        smoothed_flux = np.convolve(base_flux, kernel, mode="same")
        residuals = base_flux - smoothed_flux

        local_rms = float(np.sqrt(np.mean(residuals**2)))
        median_err = float(np.median(err_flux))
        scale_factor = local_rms / median_err if median_err > 0 else 1.0

        DevState.emit("mid_boxcar_calculation", locals())

        final_errors: Dict[str, Tuple[float, float, float]] = {}
        for name, param in winning_result.params.items():
            best_val = float(param.value)
            if param.stderr is not None and np.isfinite(param.stderr):
                scaled_err = float(param.stderr) * scale_factor
                final_errors[name] = (
                    best_val - scaled_err,
                    best_val,
                    best_val + scaled_err,
                )
            else:
                final_errors[name] = (np.nan, best_val, np.nan)

        DevState.emit("post_boxcar_calculation", locals())
        return final_errors
"""
line_models.py

Relative velocity-space Gaussian component factories for ICARUS v3.4.
Parameterizes all emission transitions exclusively in relative velocity space
(v_offset and sigma_v in km/s) and enforces strict roadmap kinematic boundaries.
"""

__project__ = "ICARUS"
__author__ = "Joseph Havens"
__version__ = "3.4.0"

from typing import Any, Optional

from astropy import constants as const
from astropy import units as u
from lmfit.models import GaussianModel

from ..core.containers import SpectralComponent
from ..core.dev_tools import DevState, dev_hook
from ..core.logger import get_icarus_logger

log = get_icarus_logger()
C_KMS = const.c.to("km/s").value

# High-ionization UV lines subject to strong radiation-driven winds
UV_WIND_IONS = (
    "C_IV_1548",
    "C_IV_1551",
    "He_II_1640",
    "Si_IV_1397",
    "O_IV_1402",
    "N_V_1240",
)


class LineComponentFactory:
    """Constructs velocity-parameterized Gaussian LMFIT components."""

    @staticmethod
    @dev_hook(step_name="build_emission_line")
    def build_emission_line(
        name: str,
        rest_wavelength: float,
        is_broad: bool = False,
        amplitude_guess: float = 1e-15,
        config: Optional[Any] = None,
        velocity_bound_kms: float = 350.0,
        sigma_init_kms: float = 120.0,
    ) -> SpectralComponent:
        """
        Builds a narrow or broad emission-line Gaussian model locked in velocity space:
          center = rest_wavelength * (1.0 + v_offset / C_KMS)
          sigma  = rest_wavelength * (sigma_v / C_KMS)
        """
        DevState.emit("pre_build_emission_line", locals())

        prefix = f"{name}_"
        model = GaussianModel(prefix=prefix)
        params = model.make_params()

        if is_broad:
            # Expand velocity offset window for high-ionization UV wind transitions
            is_uv_wind = any(uv in name for uv in UV_WIND_IONS)
            v_min = -4000.0 if is_uv_wind else -600.0
            v_max = 1000.0 if is_uv_wind else 600.0

            b_min_sig = 800.0
            b_max_sig = 8000.0
            if config is not None and getattr(config, "broad_kinematics", None):
                bk = config.broad_kinematics
                if "min_sigma" in bk and bk["min_sigma"] is not None:
                    b_min_sig = float(u.Quantity(bk["min_sigma"]).to(u.km / u.s).value)
                if "max_sigma" in bk and bk["max_sigma"] is not None:
                    b_max_sig = float(u.Quantity(bk["max_sigma"]).to(u.km / u.s).value)

            params.add(f"{prefix}v_offset", value=0.0, min=v_min, max=v_max)
            params.add(
                f"{prefix}sigma_v",
                value=max(b_min_sig * 1.5, 1200.0),
                min=b_min_sig,
                max=b_max_sig,
            )
            comp_type = "emission_broad"
        else:
            n_min_sig = 30.0
            n_max_sig = 450.0
            if config is not None:
                if getattr(config, "min_sigma", None) is not None:
                    n_min_sig = float(u.Quantity(config.min_sigma).to(u.km / u.s).value)
                if getattr(config, "max_sigma", None) is not None:
                    n_max_sig = float(u.Quantity(config.max_sigma).to(u.km / u.s).value)

            params.add(
                f"{prefix}v_offset",
                value=0.0,
                min=-velocity_bound_kms,
                max=velocity_bound_kms,
            )
            params.add(
                f"{prefix}sigma_v",
                value=min(max(sigma_init_kms, n_min_sig * 1.2), n_max_sig * 0.8),
                min=n_min_sig,
                max=n_max_sig,
            )
            comp_type = "emission_narrow"

        params[f"{prefix}center"].set(
            expr=f"{rest_wavelength:.6f} * (1.0 + {prefix}v_offset / {C_KMS:.6f})"
        )
        params[f"{prefix}sigma"].set(
            expr=f"{rest_wavelength:.6f} * ({prefix}sigma_v / {C_KMS:.6f})"
        )
        params[f"{prefix}amplitude"].set(
            value=max(float(amplitude_guess), 1e-15), min=0.0
        )

        comp = SpectralComponent(
            name=name,
            model=model,
            params=params,
            component_type=comp_type,
            rest_wavelength=rest_wavelength,
        )
        DevState.emit("post_build_emission_line", locals())
        return comp

    @staticmethod
    @dev_hook(step_name="build_outflow_line")
    def build_outflow_line(
        name: str,
        rest_wavelength: float,
        amplitude_guess: float = 1e-15,
    ) -> SpectralComponent:
        """
        Builds an asymmetric, blue-shifted ionized gas wind component for primary
        forbidden transitions ([O III] 4959/5007).
        Strictly enforces v_offset in [-2000, -50] km/s and sigma_v in [250, 765] km/s
        (corresponding to FWHM ~ 600 - 1800 km/s per Roadmap Section 1.2).
        """
        DevState.emit("pre_build_outflow_line", locals())

        prefix = f"{name}_"
        model = GaussianModel(prefix=prefix)
        params = model.make_params()

        params.add(f"{prefix}v_offset", value=-350.0, min=-2000.0, max=-50.0)
        params.add(f"{prefix}sigma_v", value=400.0, min=250.0, max=765.0)

        params[f"{prefix}center"].set(
            expr=f"{rest_wavelength:.6f} * (1.0 + {prefix}v_offset / {C_KMS:.6f})"
        )
        params[f"{prefix}sigma"].set(
            expr=f"{rest_wavelength:.6f} * ({prefix}sigma_v / {C_KMS:.6f})"
        )
        params[f"{prefix}amplitude"].set(
            value=max(float(amplitude_guess), 1e-15), min=0.0
        )

        comp = SpectralComponent(
            name=name,
            model=model,
            params=params,
            component_type="emission_outflow",
            rest_wavelength=rest_wavelength,
        )
        DevState.emit("post_build_outflow_line", locals())
        return comp
