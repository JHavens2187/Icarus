@dataclass(frozen=True, kw_only=True)
class IcarusResult:
    """
    Immutable result container returned by AGNFitter.fit().
    Holds the input spectrum, converged models, HDU tables, demographics,
    and confidence bitmask.
    """

    spectrum: Spectrum1D
    config: IcarusConfig
    continuum_eval: Optional[np.ndarray] = None
    feii_eval: Optional[np.ndarray] = None
    complex_fits: List[Tuple[LineComplex, Spectrum1D, Any]] = field(
        default_factory=list
    )
    raw_params: Dict[str, Any] = field(default_factory=dict)
    metrics: Dict[str, Any] = field(default_factory=dict)
    line_table: Optional[QTable] = None
    synth_table: Optional[QTable] = None
    confidence_flag: int = 0

    def plot(
        self, save_path: Optional[Union[str, Path]] = None, show: bool = False
    ) -> Optional[Path]:
        """Renders the 5-panel diagnostic PDF via DiagnosticPlotter."""
        out_path = Path(save_path) if save_path is not None else None
        return DiagnosticPlotter.plot_target_summary(self, out_path, show=show)

    def to_fits(self, save_path: Union[str, Path]) -> Path:
        """Writes the 3-HDU Multi-Extension FITS file via ResultWriter."""
        out_path = Path(save_path)
        return ResultWriter.write_mef_fits(self, out_path)

    def to_ecsv(self, save_path: Union[str, Path]) -> Path:
        """Writes the flat 1-row target catalog (.ecsv) for survey aggregation."""
        out_path = Path(save_path)
        return ResultWriter.write_flat_ecsv(self, out_path)

    def summary(self) -> str:
        """Returns an executive ASCII summary for terminal/Jupyter inspection."""
        t_id = self.spectrum.target.target_id
        z = self.spectrum.target.redshift
        mbh = self.metrics.get("M_BH_Msun", np.nan)
        mbh_cal = self.metrics.get("M_BH_Calibrator", "None")
        lbol = self.metrics.get("L_bol_erg_s", np.nan)
        edd = self.metrics.get("lambda_Edd", np.nan)
        sfr = self.metrics.get("SFR_uncorrected_Msun_yr", np.nan)

        lines = [
            f"================ ICARUS v{__version__} FIT SUMMARY ================",
            f"Target ID       : {t_id} (z = {z:.4f})",
            f"Complexes Fit   : {len(self.complex_fits)}",
            (
                f"log10(M_BH/Msun): {np.log10(mbh):.2f} [{mbh_cal}]"
                if (np.isfinite(mbh) and mbh > 0)
                else "log10(M_BH/Msun): N/A"
            ),
            (
                f"log10(L_bol)    : {np.log10(lbol):.2f} erg/s"
                if (np.isfinite(lbol) and lbol > 0)
                else "log10(L_bol)    : N/A"
            ),
            (
                f"Eddington Ratio : {edd:.3f}"
                if np.isfinite(edd)
                else "Eddington Ratio : N/A"
            ),
            (
                f"SFR (Msun/yr)   : {sfr:.2f}"
                if np.isfinite(sfr)
                else "SFR (Msun/yr)   : N/A"
            ),
            f"Confidence Flag : {self.confidence_flag} (0b{self.confidence_flag:06b})",
            "==============================================================",
        ]
        return "\n".join(lines)


class AGNFitter:
    """
    Stateless, picklable orchestrator class for single-target AGN spectral fitting.
    Accepts either an instantiated `Spectrum1D` or a direct FITS filepath.
    """

    BROAD_BASES = (
        "Ha",
        "H_alpha",
        "Hb",
        "H_beta",
        "Hg",
        "H_gamma",
        "Hd",
        "H_delta",
        "Heps",
        "H_epsilon",
        "Hzeta",
        "H_zeta",
        "Paa",
        "Pa_alpha",
        "Pab",
        "Pa_beta",
        "Pag",
        "Pa_gamma",
        "Pad",
        "Pa_delta",
        "Pa_epsilon",
        "Bra",
        "Br_alpha",
        "Brb",
        "Br_beta",
        "Brg",
        "Br_gamma",
        "Brd",
        "Br_delta",
        "Pfa",
        "Pf_alpha",
        "Pfb",
        "Pf_beta",
        "Pfg",
        "Pf_gamma",
        "Hua",
        "Hu_alpha",
        "Hub",
        "Hu_beta",
        "Hug",
        "Hu_gamma",
        "Lya",
        "Ly_alpha",
        "Lyb",
        "Ly_beta",
        "C_IV",
        "Mg_II",
        "He_II",
        "He_I",
        "C_III",
    )

    def __init__(
        self,
        spectrum: Union[Spectrum1D, str, Path],
        preset: Union[str, Path, IcarusConfig] = "jwst_nirspec_medium",
        redshift: Optional[float] = None,
        target_id: Optional[str] = None,
        setup: Optional[str] = None,
        column_mapping: Optional[Dict[str, str]] = None,
        **overrides: Any,
    ) -> None:
        if isinstance(preset, IcarusConfig):
            self.master_config = preset.apply_overrides(
                column_mapping=column_mapping, **overrides
            )
        else:
            self.master_config = IcarusConfig.from_preset(
                preset, column_mapping=column_mapping, **overrides
            )

        if isinstance(spectrum, (str, Path)):
            self.spectrum = Spectrum1D.read(
                filepath=spectrum,
                redshift=float(redshift or 0.0),
                target_id=target_id,
                preset=self.master_config,
                setup=setup,
                column_mapping=column_mapping,
            )
        else:
            self.spectrum = spectrum

        self.local_config = self.master_config.get_target_config(self.spectrum.target)

    @dev_hook(step_name="agn_fitter_fit")
    def fit(self) -> IcarusResult:
        """Executes the two-stage continuum and kinematic decomposition pipeline."""
        line_defs = self._load_line_defs()
        complexes = SpectralPreparer.group_into_complexes(
            line_defs, self.local_config.fitting
        )

        # Stage 1: Global Power-Law + Fe II Continuum Cascade
        global_cont_comp, global_cont_eval, feii_eval = self._run_stage1_continuum()

        # Stage 2: Local Complex Kinematic Decomposition
        complex_fits, merged_params = self._run_stage2_complexes(
            complexes, global_cont_comp, feii_eval
        )

        # Stage 3: Virial Demographics, BPT, SFR, and Confidence Bitmask
        metrics, conf_flag = DemographicsCalculator.compute_all(
            merged_params,
            self.spectrum,
            self.master_config,
            global_cont_eval=global_cont_eval,
            feii_eval=feii_eval,
        )
        merged_params.update(metrics)

        # Build HDU 1 (Line Catalog) & HDU 2 (Synthetic Spectrum) QTables
        line_table = ResultWriter.build_line_catalog_table(
            self.spectrum, complex_fits, merged_params
        )
        synth_table = ResultWriter.build_synthetic_spectrum_table(
            self.spectrum, global_cont_eval, feii_eval, complex_fits
        )

        res = IcarusResult(
            spectrum=self.spectrum,
            config=self.master_config,
            continuum_eval=global_cont_eval,
            feii_eval=feii_eval,
            complex_fits=complex_fits,
            raw_params=merged_params,
            metrics=metrics,
            line_table=line_table,
            synth_table=synth_table,
            confidence_flag=conf_flag,
        )
        DevState.flush_telemetry()
        return res

    def _load_line_defs(self) -> List[EmissionLineDef]:
        """Loads the vacuum master line table from icarus_pkg/data/."""
        line_table_path = Path(self.local_config.line_table)
        if not line_table_path.exists():
            raise FileNotFoundError(f"Master line table not found: {line_table_path}")

        line_catalog = QTable.read(line_table_path)
        str_cols = [
            c
            for c in line_catalog.colnames
            if line_catalog[c].dtype.kind in ["U", "S", "O"]
        ]
        name_col = str_cols[0]
        wave_col = [
            c for c in line_catalog.colnames if line_catalog[c].dtype.kind in ["f", "i"]
        ][0]

        profile_col = None
        for sc in str_cols[1:]:
            sample_vals = " ".join(str(v).lower() for v in line_catalog[sc][:5])
            if "narrow" in sample_vals or "broad" in sample_vals:
                profile_col = sc
                break

        target_lines = self.local_config.lines
        valid_lines = (
            line_catalog[name_col].tolist() if target_lines == "all" else target_lines
        )

        line_defs: List[EmissionLineDef] = []
        for raw_name in valid_lines:
            row_mask = line_catalog[name_col] == raw_name
            if not np.any(row_mask):
                continue

            raw_wave = line_catalog[wave_col][row_mask][0]
            rest_wave = float(
                raw_wave.value if hasattr(raw_wave, "value") else raw_wave
            )
            safe_name = sanitize_line_name(raw_name)

            if profile_col is not None:
                prof_str = str(line_catalog[profile_col][row_mask][0]).lower()
                is_broad = "broad" in prof_str and "O_III" not in safe_name
            else:
                is_broad = any(safe_name.startswith(b) for b in self.BROAD_BASES) and (
                    "O_III" not in safe_name
                )

            is_outflow = safe_name in ("O_III_4959", "O_III_5007")
            line_defs.append(
                EmissionLineDef(
                    name=safe_name,
                    rest_wavelength=rest_wave,
                    test_broad=is_broad,
                    outflow_candidate=is_outflow,
                )
            )

        return line_defs

    def _run_stage1_continuum(
        self,
    ) -> Tuple[Optional[Any], Optional[np.ndarray], Optional[np.ndarray]]:
        """Runs Stage 1 Two-Pass Global Continuum + Fe II Template Cascade."""
        return ContinuumFitter.fit_global_sed(self.spectrum, self.local_config.fitting)

    def _run_stage2_complexes(
        self,
        complexes: List[LineComplex],
        global_cont_comp: Optional[Any],
        feii_eval: Optional[np.ndarray],
    ) -> Tuple[List[Tuple[LineComplex, Spectrum1D, Any]], Dict[str, Any]]:
        """Runs Stage 2 Local Kinematic decomposition across all in-bounds complexes."""
        target = self.spectrum.target
        wave_full, _, _ = self.spectrum.to_ndarrays()
        has_feii = (
            feii_eval is not None
            and len(feii_eval) == len(wave_full)
            and np.nanmax(feii_eval) > 0
        )

        complex_fits: List[Tuple[LineComplex, Spectrum1D, Any]] = []
        merged_params: Dict[str, Any] = {
            "TargetID": target.target_id,
            "Redshift": target.redshift,
        }

        for comp in complexes:
            try:
                sliced_spec = SpectralPreparer.slice_spectrum(self.spectrum, comp)
                x_slice, y_data, _ = sliced_spec.to_ndarrays()

                fe_slice = (
                    np.interp(x_slice, wave_full, feii_eval, left=0.0, right=0.0)
                    if has_feii
                    else np.zeros_like(x_slice)
                )

                DevState.emit("pre_evaluate_broad_line", locals())

                winning_fit = CoreFitter.evaluate_broad_line(
                    comp,
                    sliced_spec,
                    self.local_config.fitting,
                    continuum=global_cont_comp,
                    feii_eval=fe_slice,
                )

                errors = CoreFitter.estimate_errors_boxcar(
                    sliced_spec, winning_fit, boxcar_width=5
                )
                merged_params[f"{comp.name}_Reduced_ChiSq"] = winning_fit.redchi

                for param_name, best_val in winning_fit.params.valuesdict().items():
                    err_16, err_50, err_84 = errors.get(
                        param_name, (np.nan, np.nan, np.nan)
                    )
                    merged_params[param_name] = best_val
                    merged_params[f"{param_name}_err16"] = err_16
                    merged_params[f"{param_name}_err50"] = err_50
                    merged_params[f"{param_name}_err84"] = err_84

                residuals = (y_data - fe_slice) - winning_fit.best_fit
                local_rms = float(
                    np.percentile(np.abs(residuals - np.median(residuals)), 68)
                )

                for line in comp.lines:
                    cont_at_line_dict = winning_fit.eval_components(
                        x=np.array([line.rest_wavelength])
                    )
                    fe_at_line = (
                        float(np.interp(line.rest_wavelength, wave_full, feii_eval))
                        if has_feii
                        else 0.0
                    )
                    cont_at_line = (
                        float(cont_at_line_dict.get("cont_", [0.0])[0]) + fe_at_line
                    )
                    merged_params[f"{line.name}_continuum_flux"] = cont_at_line

                    amp_key = f"{line.name}_narrow_amplitude"
                    if amp_key in merged_params:
                        best_amp = merged_params[amp_key]
                        inst_sig_kms = self.spectrum.get_inst_sigma_kms(
                            line.rest_wavelength
                        )
                        sigma_inst_aa = line.rest_wavelength * (
                            inst_sig_kms / 299792.458
                        )
                        f_err_approx = local_rms * SQRT_2_PI * max(sigma_inst_aa, 0.5)

                        snr = best_amp / f_err_approx if f_err_approx > 0 else 0.0
                        merged_params[f"{line.name}_narrow_snr"] = float(snr)

                        if (
                            snr < self.local_config.fitting.snr_threshold
                            or np.isnan(snr)
                            or best_amp <= 0
                        ):
                            merged_params[amp_key] = 3.0 * f_err_approx
                            merged_params[f"{line.name}_narrow_is_upper_limit"] = True
                        else:
                            merged_params[f"{line.name}_narrow_is_upper_limit"] = False

                complex_fits.append((comp, sliced_spec, winning_fit))
                DevState.emit("post_complex_success", locals())

            except IcarusDataDimensionError:
                continue
            except IcarusConvergenceError as e:
                log.warning(
                    f"[{target.target_id}] Fit failed for {comp.name}: {e}. Computing 3-sigma limits."
                )
                _, y_raw, _ = sliced_spec.to_ndarrays()
                _, _, std_dev = sigma_clipped_stats(y_raw, sigma=3.0, maxiters=5)
                for line in comp.lines:
                    inst_sig_kms = self.spectrum.get_inst_sigma_kms(
                        line.rest_wavelength
                    )
                    sigma_inst_aa = line.rest_wavelength * (inst_sig_kms / 299792.458)
                    f_err_approx = std_dev * SQRT_2_PI * max(sigma_inst_aa, 0.5)
                    merged_params[f"{line.name}_narrow_amplitude"] = 3.0 * f_err_approx
                    merged_params[f"{line.name}_narrow_is_upper_limit"] = True
                    merged_params[f"{line.name}_narrow_snr"] = 0.0
                continue
            except Exception as e:
                log.warning(
                    f"[{target.target_id}] Unexpected error in {comp.name}: {e}"
                )
                continue

        return complex_fits, merged_params

def _prompt_existing_path(prompt_text: str, is_dir: bool = False) -> Path:
    """Prompts the user for a filesystem path and validates its existence in real time."""
    while True:
        raw_val = click.prompt(
            f"  {Fore.CYAN}➤{Style.RESET_ALL} {prompt_text}", type=str
        ).strip()
        # Strip accidental surrounding quotes from drag-and-drop terminal paths
        cleaned = raw_val.strip("'").strip('"')
        candidate = Path(cleaned).expanduser()

        if not candidate.exists():
            print(f"    {Fore.RED}✖ Path does not exist: {candidate}{Style.RESET_ALL}")
            continue
        if is_dir and not candidate.is_dir():
            print(
                f"    {Fore.RED}✖ Expected a directory, got a file: {candidate}{Style.RESET_ALL}"
            )
            continue
        if not is_dir and candidate.is_dir():
            print(
                f"    {Fore.RED}✖ Expected a file, got a directory: {candidate}{Style.RESET_ALL}"
            )
            continue
        return candidate.resolve()


def _run_interactive_wizard() -> Dict[str, Any]:

    Interactive CLI configuration wizard triggered when `icarus` is invoked
    with zero command-line arguments.
    """

        f"  {Fore.MAGENTA}{Style.BRIGHT}✦ INTERACTIVE LAUNCH WIZARD ENGAGED ✦{Style.RESET_ALL}"
    )
    print(
        f"  {Fore.WHITE}{Style.DIM}No CLI flags detected. Answer the prompts below (or press Ctrl+C to abort).{Style.RESET_ALL}\n"
    )

    # 1. Execution Mode Selection
    print(f"  {Style.BRIGHT}1. Select Execution Mode:{Style.NORMAL}")
    print(
        f"     {Fore.CYAN}[1]{Style.RESET_ALL} Single FITS Spectrum {Style.DIM}(Quick Analysis / Tutorial){Style.RESET_ALL}"

        f"  {Fore.CYAN}➤{Style.RESET_ALL} Mode [1/2]",

def _process_single_target_worker(
    target: Target,
    master_config: IcarusConfig,
    data_dir: Path,
    out_dir: Path,
    dry_run: bool = False,
    generate_plots: bool = False,
    explicit_file: Optional[Path] = None,
) -> Dict[str, Any]:
    """Stateless top-level worker function for ProcessPoolExecutor pickling."""
    status: Dict[str, Any] = {
        "target_id": target.target_id,
        "status": "FAILED",
        "reason": "",
        "complexes_fit": 0,
        "ecsv_path": None,
        "summary_text": None,
    }
    try:
        local_config = master_config.get_target_config(target)
        builder = ObservationBuilder(data_dir)
        filepath = explicit_file if explicit_file else builder.locate_file(target)
        if not filepath or not filepath.exists():
            raise FileNotFoundError(
                f"Could not locate FITS file for {target.target_id}"
            )

        spectrum = builder.build_spectrum(
            target, filepath, local_config, dry_run=dry_run
        )
        if dry_run:
            status["status"] = "SUCCESS (DRY RUN)"
            return status

        fitter = AGNFitter(spectrum, preset=master_config)
        result = fitter.fit()

        if len(result.complex_fits) > 0:
            fits_path = out_dir / f"icarus_fit_{target.target_id}.fits"
            ecsv_path = out_dir / f"icarus_fit_{target.target_id}.ecsv"
            result.to_fits(fits_path)
            result.to_ecsv(ecsv_path)
            if generate_plots:
                result.plot(out_dir / f"summary_{target.target_id}.pdf")

            status["status"] = "SUCCESS"
            status["complexes_fit"] = len(result.complex_fits)
            status["ecsv_path"] = str(ecsv_path)
            status["summary_text"] = result.summary()
        else:
            status["reason"] = "Zero complexes converged."
    except Exception as e:
        log.error(f"[{target.target_id}] Worker failure: {e}")
        status["reason"] = str(e)

    return status


def aggregate_survey_catalog(
    results: List[Dict[str, Any]], out_dir: Path
) -> Optional[Path]:
    """Combines individual 1-row target ECSV files into a master survey catalog."""
    ecsv_files = [
        Path(r["ecsv_path"])
        for r in results
        if r.get("ecsv_path") and Path(r["ecsv_path"]).exists()
    ]
    if not ecsv_files:
        return None

    try:
        with warnings.catch_warnings():
            warnings.simplefilter("ignore", category=AstropyWarning)
            tables = [QTable.read(p, format="ascii.ecsv") for p in ecsv_files]
            master_table = vstack(tables, metadata_conflicts="silent")
            master_path = out_dir / "ICARUS_Master_Survey_Catalog.ecsv"
            master_table.write(master_path, format="ascii.ecsv", overwrite=True)
        return master_path
    except Exception as e:
        log.warning(f"Survey catalog aggregation warning: {e}")
        return None


@click.command()
@click.option(
    "--catalog",
    default=None,
    type=click.Path(exists=True),
    help="Path to master CSV/FITS catalog (for batch survey mode).",
)
@click.option(
    "--data",
    default=None,
    type=click.Path(exists=True),
    help="Root directory of spectral FITS files (for batch survey mode).",
)
@click.option(
    "--file",
    "single_file",
    default=None,
    type=click.Path(exists=True),
    help="Path to a single FITS file (bypasses --catalog requirement).",
)
@click.option(
    "--redshift",
    default=0.0,
    type=float,
    help="Target redshift when running in single-file mode (--file).",
)
@click.option(
    "--target-id",
    default=None,
    type=str,
    help="Optional Target ID override when running in single-file mode.",
)
@click.option(
    "--preset",
    "--config",
    "preset",
    default="jwst_nirspec_medium",
    help="Built-in preset name or path to custom YAML config.",
)
@click.option(
    "--output", default="./ICARUS_RESULTS", type=click.Path(), help="Output directory."
)
@click.option(
    "--cores",
    default=max(1, (os.cpu_count() or 2) - 1),
    type=int,
    help="CPU threads to utilize.",
)
@click.option("--plot", is_flag=True, help="Generate 5-panel PDF diagnostic plots.")
@click.option("--dry-run", is_flag=True, help="Validate inputs without executing fits.")
@click.option("--verbose", is_flag=True, help="Enable debug logging.")
@click.option(
    "--dev", type=click.Path(exists=True), help="Path to .dev breakpoints file."
)
def pipeline(
    catalog,
    data,
    single_file,
    redshift,
    target_id,
    preset,
    output,
    cores,
    plot,
    dry_run,
    verbose,
    dev,
):
    """ICARUS v3.4: AGN Kinematic Minimization & Demographics Pipeline."""
    print_banner()

    # Launch interactive wizard if invoked bare (`icarus` with no flags)
    if len(sys.argv) == 1:
        try:
            wiz = _run_interactive_wizard()
        except (KeyboardInterrupt, click.Abort):
            print(
                f"\n{Fore.YELLOW}⚠ Interactive wizard aborted by user.{Style.RESET_ALL}"
            )
            sys.exit(0)

        catalog = wiz["catalog"]
        data = wiz["data"]
        single_file = wiz["single_file"]
        redshift = wiz["redshift"]
        target_id = wiz["target_id"]
        preset = wiz["preset"]
        output = wiz["output"]
        cores = wiz["cores"]
        plot = wiz["plot"]
        dry_run = wiz["dry_run"]
        verbose = wiz["verbose"]
        dev = wiz["dev"]

    start_time = time.time()
    out_dir = Path(output)
    out_dir.mkdir(parents=True, exist_ok=True)

    if verbose:
        log.setLevel(logging.DEBUG)
        for handler in log.handlers:
            handler.setLevel(logging.DEBUG)

    if dev:
        print(
            f"{Fore.RED}{Style.BRIGHT}⚠ DEV MODE ENGAGED: Loading breakpoints from {dev}{Style.RESET_ALL}"
        )
        print(
            f"{Fore.RED}{Style.BRIGHT}⚠ OVERRIDING MULTIPROCESSING: Forcing synchronous execution for terminal access.{Style.RESET_ALL}\n"
        )
        DevState.initialize(Path(dev))
        DevState.run_preflight()
        cores = 1

    print(f"{Fore.BLUE}➤{Style.RESET_ALL} Initializing Research Environment...")
    try:
        master_config = IcarusConfig.from_preset(preset)
        log.info(f"Loaded configuration preset: {preset}")
    except IcarusConfigurationError as e:
        print(f"{Fore.RED}✖ CONFIG ERROR: {e}")
        sys.exit(1)

    explicit_file_path: Optional[Path] = None
    if single_file:
        explicit_file_path = Path(single_file)
        data_dir = explicit_file_path.parent
        tid = target_id or explicit_file_path.stem
        targets = [Target(target_id=str(tid), redshift=float(redshift))]
        cores = 1
    elif catalog and data:
        print(f"{Fore.BLUE}➤{Style.RESET_ALL} Registering Survey Catalog...")
        data_dir = Path(data)
        try:
            registry = CatalogRegistry(catalog)
            targets = registry.generate_targets()
        except Exception as e:
            print(f"{Fore.RED}✖ CATALOG ERROR: {e}")
            sys.exit(1)
    else:
        print(
            f"{Fore.RED}✖ USAGE ERROR: Provide either (--catalog and --data) for batch mode, (--file [--redshift]) for single-target mode, or run bare with zero flags to launch the interactive wizard."
        )
        sys.exit(1)

    num_targets = len(targets)
    mode_str = (
        f"{Fore.YELLOW}DRY RUN (Validation Mode)"
        if dry_run
        else (
            f"{Fore.GREEN}SINGLE-TARGET FIT"
            if single_file
            else f"{Fore.GREEN}BATCH SURVEY PRODUCTION"
        )
    )
    manifest = f"""
    {Style.BRIGHT}--- PIPELINE MANIFEST ---{Style.NORMAL}
    {Fore.WHITE}Mode       : {mode_str}
    {Fore.WHITE}Preset     : {Fore.CYAN}{preset}
    {Fore.WHITE}Threads    : {Fore.CYAN}{cores} Core{'s' if cores > 1 else ''} {Fore.RED + '(DEV OVERRIDE)' if dev else ''}
    {Fore.WHITE}Sources    : {Fore.CYAN}{num_targets} Target{'s' if num_targets != 1 else ''}
    {Fore.WHITE}Output     : {Fore.CYAN}{out_dir.resolve()}
    {Fore.WHITE}Plots      : {Fore.CYAN}{'Enabled (5-Panel PDFs)' if plot else 'Disabled'}
    """
    print(manifest)

    results: List[Dict[str, Any]] = []
    success_count = 0
    total_complexes_fit = 0

    if cores == 1:
        print(f"{Fore.MAGENTA}{Style.BRIGHT}🚀 ENGAGING SYNCHRONOUS EXECUTION LOOP\n")
        for i, t in enumerate(targets):
            res = _process_single_target_worker(
                t,
                master_config,
                data_dir,
                out_dir,
                dry_run,
                plot,
                explicit_file=explicit_file_path,
            )
            results.append(res)
            progress = (i + 1) / num_targets * 100
            if "SUCCESS" in res["status"]:
                success_count += 1
                status_ind = f"{Fore.GREEN}✔"
                total_complexes_fit += res.get("complexes_fit", 0)
            else:
                status_ind = f"{Fore.RED}✘"

            short_reason = (
                (res["reason"][:40] + "...")
                if len(res.get("reason", "")) > 40
                else res.get("reason", "")
            )
            print(
                f" {status_ind} [{progress:4.1f}%] {Fore.WHITE}{t.target_id:<15} {Style.DIM}| {res['status']:<10} {short_reason}",
                end="\r",
            )
            sys.stdout.flush()
    else:
        print(f"{Fore.MAGENTA}{Style.BRIGHT}🚀 ENGAGING MULTIPROCESSING POOL\n")
        with ProcessPoolExecutor(max_workers=cores) as executor:
            futures = {
                executor.submit(
                    _process_single_target_worker,
                    t,
                    master_config,
                    data_dir,
                    out_dir,
                    dry_run,
                    plot,
                ): t
                for t in targets
            }
            for i, future in enumerate(as_completed(futures)):
                t_obj = futures[future]
                res = future.result()
                results.append(res)
                progress = (i + 1) / num_targets * 100
                if "SUCCESS" in res["status"]:
                    success_count += 1
                    status_ind = f"{Fore.GREEN}✔"
                    total_complexes_fit += res.get("complexes_fit", 0)
                else:
                    status_ind = f"{Fore.RED}✘"

                short_reason = (
                    (res["reason"][:40] + "...")
                    if len(res.get("reason", "")) > 40
                    else res.get("reason", "")
                )
                print(
                    f" {status_ind} [{progress:4.1f}%] {Fore.WHITE}{t_obj.target_id:<15} {Style.DIM}| {res['status']:<10} {short_reason}",
                    end="\r",
                )
                sys.stdout.flush()

    duration = time.time() - start_time
    df_report = pd.DataFrame(
        [{k: v for k, v in r.items() if k != "summary_text"} for r in results]
    )
    report_path = out_dir / "ICARUS_Execution_Report.csv"
    df_report.to_csv(report_path, index=False)

    master_cat_path = None
    if num_targets > 1 and not dry_run:
        master_cat_path = aggregate_survey_catalog(results, out_dir)

    print(
        f"\n\n{Fore.CYAN}{Style.BRIGHT}=================================================="
    )
    print(
        f"{Fore.CYAN}{Style.BRIGHT}               SURVEY DEBRIEFING                  "
    )
    print(
        f"{Fore.CYAN}{Style.BRIGHT}=================================================="
    )

    if single_file and results and results[0].get("summary_text"):
        print(f"\n{results[0]['summary_text']}")

    summary_stats = f"""
    {Fore.WHITE}Total Runtime   : {Fore.YELLOW}{duration:.2f} seconds
    {Fore.WHITE}Average Pace    : {Fore.YELLOW}{duration / max(num_targets, 1):.2f} s/target
    {Fore.WHITE}Execution Log   : {Fore.CYAN}{report_path.name}
    {Fore.WHITE}Survey Catalog  : {Fore.CYAN}{master_cat_path.name if master_cat_path else 'N/A (Single Target)'}

    {Style.BRIGHT}--- SCIENCE YIELD ---{Style.NORMAL}
    {Fore.WHITE}Target Success  : {Fore.GREEN}{success_count} {Fore.WHITE}/ {num_targets}
    {Fore.WHITE}Complexes Fit   : {Fore.GREEN}{total_complexes_fit}
    """
    print(summary_stats)

    if success_count == num_targets:
        print(
            f"{Fore.GREEN}{Style.BRIGHT}✔ PIPELINE EXECUTION COMPLETE. 100% TARGET CONVERGENCE.{Style.RESET_ALL}"
        )
    else:
        print(
            f"{Fore.YELLOW}{Style.BRIGHT}⚠ PIPELINE EXECUTION COMPLETE WITH WARNINGS. SEE REPORT FOR DETAILS.{Style.RESET_ALL}"
        )


if __name__ == "__main__":
    pipeline()
